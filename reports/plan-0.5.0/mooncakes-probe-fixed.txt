import json
import os
from pathlib import Path
import subprocess
import tempfile
import threading
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer

ROOT = Path('/Users/cjy/Documents/ChatGPT/黑客松新项目/projects/visionleak')
requests = []
status = 202

class Handler(BaseHTTPRequestHandler):
    def do_POST(self):
        length = int(self.headers.get('Content-Length', '0'))
        self.rfile.read(length)
        requests.append({'path': self.path, 'bytes': length})
        detail = 'Dry run completed successfully. No changes were made. The dry-run was made for package localtest/dryrun version 0.1.0.'
        code = status if self.path.startswith('/api/v0/publish') else 200
        body = json.dumps({'detail': detail, 'ok': True, 'username': 'localtest'}).encode()
        self.send_response(code)
        self.send_header('Content-Type', 'application/json')
        self.send_header('Content-Length', str(len(body)))
        self.end_headers()
        self.wfile.write(body)

    def log_message(self, *args):
        pass

with tempfile.TemporaryDirectory(prefix='vl-mooncake-probe-') as tmp:
    base = Path(tmp)
    home = base / 'home'
    home.mkdir()
    for directory in ['bin', 'lib', 'include']:
        (home / directory).symlink_to(ROOT / '.toolchain' / directory)
    (home / 'credentials.json').write_text(json.dumps({'token': 'local-probe-only-not-a-real-token', 'username': 'localtest'}))
    project = base / 'project'
    project.mkdir()
    (project / 'moon.mod').write_text('name = "localtest/dryrun"\nversion = "0.1.0"\nlicense = "MIT"\n')
    (project / 'moon.pkg').write_text('')
    (project / 'main.mbt').write_text('pub fn answer() -> Int { 42 }\n')
    server = ThreadingHTTPServer(('127.0.0.1', 0), Handler)
    thread = threading.Thread(target=server.serve_forever, daemon=True)
    thread.start()
    try:
        output = []
        for status in [200, 202, 401, 500]:
            requests.clear()
            env = {**os.environ, 'MOON_HOME': str(home), 'VISIONLEAK_MOONCAKES_HOME': str(home), 'MOONCAKES_REGISTRY': f'http://127.0.0.1:{server.server_port}', 'NO_PROXY': '127.0.0.1', 'no_proxy': '127.0.0.1'}
            result = subprocess.run([str(ROOT / 'tools/moon'), 'publish', '--dry-run', '--frozen'], cwd=project, env=env, text=True, capture_output=True, timeout=30)
            row = {'status': status, 'exitCode': result.returncode, 'requests': list(requests), 'stdout': result.stdout, 'stderr': result.stderr}
            output.append(row)
            print(json.dumps(row), flush=True)
            assert result.returncode == (0 if status in (200, 202) else 255), row
            assert len(requests) == 1 and requests[0]['path'] == '/api/v0/publish', row
        (ROOT / 'reports/plan-0.5.0/mooncakes-response-probe-fixed.json').write_text(json.dumps(output, indent=2) + '\n')
    finally:
        server.shutdown()
