test "published core exact digest and default audit" {
  assert_eq(@visionleak.sha256(b"abc"), "ba7816bf8f01cfea414140de5dae2223b00361a396177a9cb410ff61f20015ad")
  assert_eq(@visionleak.hamming(String::make(64, '0'), String::make(64, '1')), 64)
  let report = @visionleak.audit_json("{\"images\":[],\"options\":{}}")
  assert_eq(report.contains("\"complete\":true"), true)
}

test "published 0.5.0 local crop export" {
  let fingerprint = @visionleak.fingerprint_local_rgba(1, 1, b"\xff\x00\x00\xff")
  assert_eq(fingerprint.contains("vl-local-dhash-v1"), true)
  assert_eq(fingerprint.contains("localHashes"), true)
}
